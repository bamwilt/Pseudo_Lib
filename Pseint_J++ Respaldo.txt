Función result <- __private_String_Indexof_general(text,text_match,isLast)
	Definir result, i, index_Text, index_Match, int_break, end_index_match Como Entero;
	Definir text_Current Como Cadena;
	Definir condition_index Como Lógico;
	index_Text <- String_length(text);
	index_Match <- String_length(text_match);
	result <- -1;
	i <- if(isLast,index_Text-index_Match,0,TYPE_INT);
	condition_index <- if(isLast,i>=0,i<=index_Text,TYPE_BOOLEAN);
	int_break <- if(isLast,-1,increment(i)+index_Text,TYPE_INT);
	end_index_match <- if(isLast,min_int(decrement(index_Match),0),index_Text-index_Match,TYPE_INT);
	Mientras condition_index Hacer
		text_Current <- String_substring(text,i,i+index_Match);
		result <- if(String_isEquals(text_Current,text_match),i,-1,TYPE_INT);
		i <- if(isLast,decrement(i),increment(i),TYPE_INT);
		i <- if(result>-1,int_break,i,TYPE_INT);
		i <- if(i==end_index_match,int_break,i,TYPE_INT); // break
		condition_index <- if(isLast,i>=0,i<=index_Text,TYPE_BOOLEAN); // break
	FinMientras
FinFunción

Función Date <- Local_Date_ofString(year,month,day)
	Definir Date, dateParts, year_format, month_format, day_format Como Cadena;
	Definir int_year, int_month, int_day Como Real;
	Dimensionar dateParts(3);
	int_year <- min_int(String_ToNum(year),0);
	int_month <- rangeLimit_Int(String_ToNum(month),1,12);
	int_day <- rangeLimit_Int(String_ToNum(day),1,31);
	year_format <- String_substring_from('0000',String_length(Num_ToString(int_year)));
	month_format <- if(String_length_index(Num_ToString(int_month))<1,'0','',TYPE_STRING); // pad agregar
	day_format <- if(String_length_index(Num_ToString(int_day))<1,'0','',TYPE_STRING);
	dateParts[0] <- String_join(year_format,Num_ToString(int_year));
	dateParts[1] <- String_join(month_format,Num_ToString(int_month));
	dateParts[2] <- String_join(day_format,Num_ToString(int_day));
	Date <- String_join_array_withSeparator(dateParts,3,'-');
FinFunción

// y ear (y o yy 25) yyy 025 yyyy 2025 yyyyy 2025
// M month name(+MMM) or number
// d ay del mes
// D dia del anio
// E nameDay E abrevabation EEEE completo
// e Daynum of semana
// W semana del mes 1-53
// w semana del anio
// a AM/PM
// 'texto' "texto"
// H hora militar
// h hora
// m minutos
// s segundos
// S miliSegundos
// 30.6m-29.4-i
// si es bisiesto i = 3 sino i = 2, si es enero/ febrero no restar si es i
Función Date <- Local_Date_format(text)
	Definir Date, dateParts, DateTemp Como Cadena;
	Definir index Como Entero;
	Dimensionar dateParts(3);
	dateParts[0] <- String_substring_from_start(Local_Date_now,4);
	dateParts[1] <- String_substring(Local_Date_now,4,6);
	dateParts[2] <- String_substring_from_end(Local_Date_now,2);
	Si String_Constains(text,'Y') Entonces
		index <- String_indexOf(text,'Y');
		Date <- String_insert(String_remove(text,index,index+1),dateParts[0],index);
	FinSi
	Si String_Constains(text,'M') Entonces
		index <- String_indexOf(Date,'M');
		Date <- String_insert(String_remove(Date,index,index+1),dateParts[1],index);
	FinSi
FinFunción
