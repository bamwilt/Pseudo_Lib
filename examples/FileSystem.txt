Funcion init_FileSystem

	name_archive = NameArchive_Defualt();
	archive_current = util_List_New(TYPE_STRING());
	archive_current = util_List_add(archive_current, " ");
	folder_archive = util_Map_New(TYPE_STRING(), TYPE_STRING());
	folder_archive = util_Map_Put(folder_archive, name_archive, archive_current);
	folder_archive = folder_setCurrentName(folder_archive, name_archive);
