		list_result = __private_util_List_OffSet_update_groupNum(list_result,num_Group);// 2 --> 4 : 0000 0000 list_group
		list_result = __private_util_List_OffSet_update_metadata_range(list_result, num_group);// 0,0,0,1 index memory

Funcion OffsetInfo_element <- __private_util_List_OffSet_GetElementOffsetInfo(struct_List, index_element)
	Definir OffsetInfo_element, subtract_carry, multiplication_carry, element_rest, num_AreaSize como Numero;
	Definir index_StartSize, index_EndSize, sum_carry, index_element_bitFormat, index_start_search como Numero;
	Definir element_Length, element_Inner_index, separator_info, i, end_carry como Numero;
	Definir AreaSizeGroup, num_AreaSize_str como Texto;
	AreaSizeGroup = String_substring(struct_List,increment(String_lastIndexOf(struct_List,symbol_Separator())),String_lastIndexOf(struct_List,symbol_Separator_Simple()));
	index_endSize = String_length(AreaSizeGroup);
	subtract_carry = 0;
	multiplication_carry = 0;
	element_rest = index_element;
	num_AreaSize = 0;
	end_carry = __private_util_List_OffSet_getLengthBit(struct_List) * 2;
	Mientras multiplication_carry < end_carry & num_AreaSize <= index_element Hacer
		multiplication_carry = increment_step(multiplication_carry, 4);
		si index_StartSize > -1  Entonces
			index_StartSize = String_lastIndexOf_fromIndex(AreaSizeGroup,symbol_Separator_Simple(),index_endSize);
			num_AreaSize_str = String_substring(AreaSizeGroup, increment(index_StartSize), index_EndSize);
			num_AreaSize = String_ToNum(num_AreaSize_str);
			index_EndSize = index_StartSize;
		FinSi
		
		si (num_AreaSize <= index_element) & num_AreaSize > 0 Entonces
			subtract_carry = decrement_step(num_AreaSize, subtract_carry);
			sum_carry = increment_step((subtract_carry * multiplication_carry), sum_carry);
			element_rest = decrement_step(element_rest, subtract_carry);
		FinSi
		subtract_carry = num_AreaSize;
	FinMientras
	index_start_search = String_lastIndexOf(struct_List,symbol_Separator());
	index_element_bitFormat = decrement_step(index_start_search,increment_step((element_rest * multiplication_carry), sum_carry));//122-((2*4)+4)
	OffsetInfo_element = increment_step((index_element_bitFormat*100),multiplication_carry);// index + multCarry_info
FinFuncion

Funcion list_result <- __private_util_List_OffSet_decrement_AreaSizeGroup_ByIndex(struct_List, index_element)
	Definir list_result, AreaSizeGroup, AreaSizeGroup_carry, AreaInfo, num_AreaSize_str como Texto;
	Definir i,subtract_carry, index_endSize, index_StartSize, num_AreaSize , index_AreaSizeGroup, index_AreaInfo, list_length como Numero;
	index_AreaInfo = String_lastIndexOf(struct_List,symbol_Separator_Simple());
	index_AreaSizeGroup = String_lastIndexOf_fromIndex(struct_list,symbol_Separator(),index_AreaInfo);
	list_length = String_length(struct_List);
	
	AreaInfo = String_substring(struct_List,increment(index_AreaInfo), list_length);
	AreaSizeGroup = String_substring(struct_List,increment(index_AreaSizeGroup),index_AreaInfo );
	index_endSize = String_length(AreaSizeGroup);
	i = 0;
	Mientras i < 4 Hacer
		index_StartSize = String_lastIndexOf_fromIndex(AreaSizeGroup,symbol_Separator_Simple(),index_endSize);
		num_AreaSize_str = String_substring(AreaSizeGroup, increment(index_StartSize), index_EndSize);
		num_AreaSize = String_ToNum(num_AreaSize_str);

		si (num_AreaSize > index_element) & num_AreaSize > 0 Entonces
			num_AreaSize_str = num_ToString(decrement(num_AreaSize));
		FinSi
		i= increment(i);
		AreaSizeGroup_carry = String_append_withSeparator(num_AreaSize_str,AreaSizeGroup_carry, symbol_Separator_Simple()); 
	FinMientras
	
	list_result = String_Delete(struct_List,increment(index_AreaSizeGroup),list_length);
